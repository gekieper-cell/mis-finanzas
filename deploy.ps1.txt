# =====================================================================
#  deploy.ps1 - Sube Mis Finanzas a GitHub y Vercel
#  Supabase ya esta configurado (esquema aplicado y verificado).
#
#  Uso (desde esta carpeta):
#     powershell -ExecutionPolicy Bypass -File .\deploy.ps1
#
#  Las claves se pegan cuando Vercel las pide: no quedan en el
#  historial de PowerShell ni en ningun archivo del repo.
# =====================================================================

$ErrorActionPreference = "Continue"   # PS 5.1: con "Stop", el stderr de git/vercel corta el script
Set-Location $PSScriptRoot

$SUPABASE_URL = "https://cfsysumtjsdofjcvocho.supabase.co"
$SUPABASE_REF = "cfsysumtjsdofjcvocho"
$VERCEL = "vercel.cmd"   # .cmd evita el bloqueo de ExecutionPolicy sobre vercel.ps1

function Paso($n, $t) { Write-Host "`n==== $n  $t ====" -ForegroundColor Cyan }
function Ok($t)   { Write-Host "  [OK] $t" -ForegroundColor Green }
function Info($t) { Write-Host "  $t" -ForegroundColor Gray }
function Falla($t) { Write-Host "  [ERROR] $t" -ForegroundColor Red; exit 1 }
function Run { param([string]$exe, [string[]]$a)
  & $exe @a
  if ($LASTEXITCODE -ne 0) { Falla "$exe $($a -join ' ') termino con codigo $LASTEXITCODE" }
}

# ---------------------------------------------------------------------
Paso "1/5" "Requisitos"
foreach ($c in "git", "node", "npm") {
  if (-not (Get-Command $c -ErrorAction SilentlyContinue)) { Falla "Falta '$c' en el PATH" }
}
Ok "git, node y npm disponibles"
if (-not (Get-Command $VERCEL -ErrorAction SilentlyContinue)) {
  Info "Instalando Vercel CLI..."
  Run "npm.cmd" @("install", "-g", "vercel")
}
Ok ("Vercel CLI " + (& $VERCEL --version 2>$null | Select-Object -Last 1))

# ---------------------------------------------------------------------
Paso "2/5" "GitHub"
$remote = git remote get-url origin 2>$null
if (-not $remote) { Falla "El repo no tiene 'origin' configurado" }
Info "Remoto: $remote"
git add -A
if (git status --porcelain) {
  Run "git" @("commit", "-m", "Actualizacion $(Get-Date -Format 'yyyy-MM-dd HH:mm')")
}
Run "git" @("push", "-u", "origin", "main")
Ok "Codigo en GitHub"

# ---------------------------------------------------------------------
Paso "3/5" "Supabase"
Ok "Esquema, RLS, funciones y usuario ya verificados - nada que ejecutar"

# ---------------------------------------------------------------------
Paso "4/5" "Vercel"
& $VERCEL whoami 2>$null | Out-Null
if ($LASTEXITCODE -ne 0) {
  Info "Inicia sesion (elegi 'Continue with GitHub')..."
  Run $VERCEL @("login")
}
Ok ("Sesion: " + (& $VERCEL whoami 2>$null | Select-Object -Last 1))

if (-not (Test-Path ".vercel\project.json")) {
  Info "Creando/vinculando el proyecto 'mis-finanzas'..."
  Run $VERCEL @("link", "--yes")
}
Ok "Proyecto vinculado"

Info "Conectando el repo de GitHub (deploy automatico en cada push)..."
& $VERCEL git connect --yes 2>&1 | Out-Host   # si ya estaba conectado, solo avisa

$envs = (& $VERCEL env ls 2>&1 | Out-String)
foreach ($target in "production", "development") {
  foreach ($name in "NEXT_PUBLIC_SUPABASE_URL", "NEXT_PUBLIC_SUPABASE_ANON_KEY") {
    $pattern = "(?m)^\s*$name\s.*\b$target\b"
    if ($envs -match $pattern) { Ok "$name ($target) ya cargada"; continue }
    Write-Host ""
    if ($name -eq "NEXT_PUBLIC_SUPABASE_URL") {
      Write-Host "  Cuando pregunte el valor de $name ($target), pega:" -ForegroundColor Yellow
      Write-Host "     $SUPABASE_URL" -ForegroundColor White
    } else {
      Write-Host "  Cuando pregunte el valor de $name ($target), pega la clave ANON o PUBLISHABLE" -ForegroundColor Yellow
      Write-Host "  (Supabase > Project Settings > API Keys). NUNCA la service_role / secret." -ForegroundColor Yellow
    }
    Run $VERCEL @("env", "add", $name, $target)
  }
}
Ok "Variables de entorno cargadas"

Info "Deploy a produccion (1-2 minutos)..."
$deployUrl = (& $VERCEL deploy --prod --yes) | Select-Object -Last 1
if ($LASTEXITCODE -ne 0 -or -not $deployUrl) { Falla "El deploy fallo. Mira los logs en vercel.com > mis-finanzas > Deployments" }
$inspect = (& $VERCEL inspect $deployUrl 2>&1 | Out-String)
$aliases = @([regex]::Matches($inspect, "https://[a-z0-9-]+\.vercel\.app") | ForEach-Object { $_.Value } | Select-Object -Unique | Sort-Object Length)
$prodUrl = if ($aliases) { $aliases[0] } else { $deployUrl }
Ok "Publicado: $prodUrl"

# ---------------------------------------------------------------------
Paso "5/5" "Verificacion"
try {
  $r = Invoke-WebRequest "$prodUrl/login" -UseBasicParsing -TimeoutSec 30
  $csp = $r.Headers["Content-Security-Policy"]
  if ($r.StatusCode -eq 200) { Ok "/login responde 200" }
  if ($csp -match [regex]::Escape($SUPABASE_URL)) { Ok "CSP apunta a tu Supabase (variables bien aplicadas)" }
  else { Write-Host "  [!] La CSP no incluye tu URL de Supabase: revisa NEXT_PUBLIC_SUPABASE_URL y redeploya" -ForegroundColor Yellow }
} catch {
  Write-Host "  [!] No pude verificar automaticamente: $($_.Exception.Message)" -ForegroundColor Yellow
  Write-Host "      Si es 401, Vercel tiene 'Deployment Protection' en esa URL; probala en el navegador." -ForegroundColor Yellow
}

Write-Host ""
Write-Host "Ultimo paso manual (una sola vez):" -ForegroundColor Cyan
Write-Host "  Supabase > Authentication > URL Configuration > Site URL = $prodUrl"
Start-Process "https://supabase.com/dashboard/project/$SUPABASE_REF/auth/url-configuration"
Start-Process $prodUrl
Write-Host "`nListo. Desde ahora, cada 'git push' redeploya solo." -ForegroundColor Green
